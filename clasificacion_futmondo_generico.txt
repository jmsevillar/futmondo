const { chromium } = require('playwright');
const fs = require('fs');

// 1. Capturar y validar argumentos de consola
const args = process.argv.slice(2);
const usuarioArgumento = args[0];
const passwordArgumento = args[1];

if (!usuarioArgumento || !passwordArgumento) {
    console.error('\n[ERROR] Faltan parámetros obligatorios.');
    console.log('Uso correcto: node semaantics.js <usuario> <contraseña>');
    console.log('Ejemplo:      node semaantics.js mi_correo@hotmail.com miContraseña123\n');
    process.exit(1);
}

async function clickFlutter(page, relativeX, relativeY, descripcion) {
    const flutter = page.locator('flutter-view');
    console.log(`[ESPERA] Buscando Flutter para: ${descripcion}`);

    await flutter.waitFor({
        state: 'visible',
        timeout: 20000
    });

    const rect = await flutter.boundingBox();
    if (!rect) {
        throw new Error('No se encuentra la vista de Flutter');
    }

    const x = rect.x + rect.width * relativeX;
    const y = rect.y + rect.height * relativeY;

    console.log(`[CLICK] ${descripcion} (X=${Math.round(x)}, Y=${Math.round(y)})`);
    await page.mouse.click(x, y);
    console.log(`[OK] ${descripcion}`);
}

(async () => {
    console.log(`\n[INICIO] Ejecutando extracción automática (Modo Visible Estable) para: ${usuarioArgumento}...`);

    // Modo visible estable
    const browser = await chromium.launch({
        headless: false
    });

    const page = await browser.newPage({
        ignoreHTTPSErrors: true
    });

    let datosGuardados = false;

    // INTERCEPTOR DE RED: Captura y procesa los datos usando tu lógica exacta
    page.on('response', async (response) => {
        const url = response.url();
        if (url.includes('/championship/teams')) {
            try {
                console.log('\n[API CAPTURADA] ¡Se han detectado los datos de los equipos!');
                const jsonResponse = await response.json();
                
                const equipos = Array.isArray(jsonResponse) ? jsonResponse : (jsonResponse.teams || jsonResponse.answer?.teams || jsonResponse.answer || []);

                if (equipos.length === 0) {
                    console.error('[ERROR] No se reconoció la lista de equipos en el JSON crudo.');
                    return;
                }

                const clasificacionLimpia = equipos.map((eq, index) => {
                    return {
                        posicion: index + 1,
                        nombreEquipo: eq.name || eq.team_name || 'Desconocido',
                        usuario: eq.user?.name || eq.username || 'Sin usuario',
                        puntos: eq.points || eq.total_points || 0,
                        valorEquipo: eq.teamValue || eq.value || 0
                    };
                });

                let usuarioUsado = 't_sevi'; 
                const primerEquipoConUsuario = equipos.find(eq => eq.user?.name || eq.username);
                if (primerEquipoConUsuario) {
                    const nombreEncontrado = primerEquipoConUsuario.user?.name || primerEquipoConUsuario.username;
                    usuarioUsado = nombreEncontrado.split('@')[0].replace(/[^a-zA-Z0-9__-]/g, '');
                }

                const hoy = new Date();
                const año = hoy.getFullYear();
                const mes = String(hoy.getMonth() + 1).padStart(2, '0');
                const dia = String(hoy.getDate()).padStart(2, '0');
                const fechaFormateada = `${año}-${mes}-${dia}`;

                const nombreArchivoFinal = `clasificacion_${usuarioUsado}_${fechaFormateada}.json`;

                const registroEstructurado = {
                    usuario_sesion: usuarioUsado,
                    fecha_captura: hoy.toISOString(),
                    datos: clasificacionLimpia
                };

                fs.writeFileSync(nombreArchivoFinal, JSON.stringify(registroEstructurado, null, 2), 'utf-8');
                console.log(`[ÉXITO] Archivo generado: "${nombreArchivoFinal}"`);
                console.log(`        Contiene ${clasificacionLimpia.length} equipos de tu comunidad.`);
                
                datosGuardados = true;
            } catch (e) {
                // Ignorar fallos de parsing en otras respuestas
            }
        }
    });

    try {
        // PASO 1: Abrir Futmondo tradicional
        console.log('\n[PASO 1] Abriendo Futmondo...');
        await page.goto('https://futmondo.com');
        console.log('[OK PASO 1] Página principal cargada');

        // PASO 2: Pulsar Jugar en la web HTML clásica
        console.log('\n[PASO 2] Pulsando Jugar de la web...');
        await page.getByRole('link', { name: '¡Jugar!', exact: true }).click();
        console.log('[OK PASO 2] Enlace pulsado');

        // PASO 3: Esperar el lienzo de Flutter con pausas de seguridad amplias
        console.log('\n[PASO 3] Esperando inicialización de Flutter...');
        await page.locator('flutter-view').waitFor({ state: 'visible' });
        await page.waitForTimeout(3000); 
        console.log('[OK PASO 3] Flutter localizado y listo');

        // PASO 4: Jugar Flutter
        console.log('\n[PASO 4] Pulsando Jugar Flutter...');
        await clickFlutter(page, 0.53008, 0.88276, 'Jugar Flutter');
        await page.waitForTimeout(3000);

        // PASO 5: Abrir login
        console.log('\n[PASO 5] Abriendo login...');
        await clickFlutter(page, 0.49688, 0.17471, 'Contenedor login');
        await page.waitForTimeout(2000); 

        // PASO 6: Escribir usuario introducido por parámetro
        console.log('\n[PASO 6] Introduciendo usuario...');
        await clickFlutter(page, 0.503, 0.274, 'Campo usuario');
        await page.waitForTimeout(500);
        await page.keyboard.type(usuarioArgumento, { delay: 50 });
        console.log('[OK PASO 6] Usuario introducido');

        // PASO 7: Escribir contraseña introducida por parámetro
        console.log('\n[PASO 7] Introduciendo contraseña...');
        await clickFlutter(page, 0.503, 0.382, 'Campo contraseña');
        await page.waitForTimeout(500);
        await page.keyboard.type(passwordArgumento, { delay: 50 });
        console.log('[OK PASO 7] Contraseña introducida');

        // PASO 8: Acceder
        console.log('\n[PASO 8] Pulsando ACCEDER...');
        await clickFlutter(page, 0.570, 0.901, 'Botón ACCEDER');
        console.log('[ESPERA] Esperando autenticación...');
        await page.waitForTimeout(8000);

        // PASO 10: Omitir preconfiguración
        console.log('\n[PASO 10] Pulsando OMITIR preconfiguración...');
        await page.waitForTimeout(2000);
        await clickFlutter(page, 0.49319, 0.89037, 'Omitir preconfiguración');
        console.log('[ESPERA] Esperando a que cargue la interfaz principal...');
        await page.waitForTimeout(6000);

        // PASO 11: Clic automático en la sección de Equipos con tus coordenadas calibradas
        console.log('\n[PASO 11] Pulsando en la sección EQUIPOS automáticamente...');
        await clickFlutter(page, 0.59219, 0.25278, 'Sección Equipos'); 

        // Espera de hasta 15 segundos para dar tiempo a recibir el JSON y guardarlo
        console.log('[ESPERA] Confirmando interceptación de red y guardado...');
        for (let i = 0; i < 15; i++) {
            if (datosGuardados) break;
            await page.waitForTimeout(1000);
        }

        if (!datosGuardados) {
            console.log('[AVISO] El proceso terminó pero no se logró capturar la respuesta de la API de equipos de forma automática.');
        }

        await browser.close();
        console.log('[FIN] Proceso terminado y navegador cerrado.\n');

    } catch (error) {
        console.error('\n[ERROR] AUTOMATIZACIÓN INTERRUMPIDA:');
        console.error(error.message);
        await browser.close();
    }
})();
